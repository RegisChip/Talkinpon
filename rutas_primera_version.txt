const MAP_W = 770;
const MAP_H = 680;

let modo = 'uno';
let origenFijoId = null;
let seleccionOrigenId = null;
let seleccionDestinoId = null;

const campoUbicacionActual = document.getElementById('campoUbicacionActual');
const campoOrigen = document.getElementById('campoOrigen');
const infoOrigenFijo = document.getElementById('infoOrigenFijo');
const infoOrigen = document.getElementById('infoOrigen');
const infoDestino = document.getElementById('infoDestino');
const infoModo = document.getElementById('infoModo');
const btnModoDos = document.getElementById('btnModoDos');
const btnModoUno = document.getElementById('btnModoUno');
const btnCalcular = document.getElementById('btnCalcular');
const btnNuevaRuta = document.getElementById('btnNuevaRuta');
const cont = document.getElementById('mapa');
const svg = document.getElementById('relaciones');
const debugLog = document.getElementById('debugLog');

function log(msg, tipo='info'){
  console.log(msg);
  const line = document.createElement('div');
  line.className = 'debug-line';
  const timestamp = new Date().toLocaleTimeString();
  const icon = tipo==='error' ? '❌' : tipo==='success' ? '✅' : tipo==='warning' ? '⚠️' : 'ℹ️';
  line.innerHTML = `[${timestamp}] ${icon} ${msg}`;
  debugLog.appendChild(line);
  debugLog.scrollTop = debugLog.scrollHeight;
}

window.addEventListener('load', () => {
  log(`Sistema iniciado. Edificios cargados: ${edificios.length}`);
  inicializar();
});

function inicializar(){
  if (Array.isArray(edificios) && edificios.length > 0){
    const edificioA = edificios.find(e => e.nombre_edificio.trim().toUpperCase() === 'A');
    if (edificioA) {
      origenFijoId = Number(edificioA.id);
      infoOrigenFijo.textContent = edificioA.nombre_edificio;
      log(`Origen fijo establecido: ${edificioA.nombre_edificio} (ID: ${origenFijoId})`);
    } else {
      origenFijoId = Number(edificios[0].id);
      infoOrigenFijo.textContent = edificios[0].nombre_edificio;
      log(`No se encontró edificio 'A', usando ${edificios[0].nombre_edificio} como origen (ID: ${origenFijoId})`, 'warning');
    }
  } else {
    infoOrigenFijo.textContent = 'No hay edificios';
    log('ERROR: No hay edificios disponibles', 'error');
  }
  configurarEventos();
  cambiarModo('uno');
}

function configurarEventos(){
  btnModoUno.addEventListener('click',()=>cambiarModo('uno'));
  btnModoDos.addEventListener('click',()=>cambiarModo('dos'));
  btnCalcular.addEventListener('click', calcularRuta);
  btnNuevaRuta.addEventListener('click',buscarNuevaRuta);
}

function cambiarModo(nuevo){
  modo = nuevo;
  btnModoUno.classList.toggle('active', modo==='uno');
  btnModoDos.classList.toggle('active', modo==='dos');
  infoModo.textContent = (modo==='uno')?'Una ubicación':'Dos ubicaciones';
  campoUbicacionActual.style.display = (modo==='uno')?'flex':'none';
  campoOrigen.style.display = (modo==='dos')?'flex':'none';
  seleccionOrigenId = null;
  seleccionDestinoId = null;
  infoOrigen.textContent = 'No seleccionado';
  infoDestino.textContent = 'No seleccionado';
  btnNuevaRuta.style.display = 'none';
  btnCalcular.style.display = 'inline-block';
  renderizarBotones();
  log(`Modo cambiado a: ${modo==='uno'?'Ver Ubicación':'Calcular Ruta'}`);
}

function renderizarBotones(){
  cont.querySelectorAll('.ubicacion-btn').forEach(b=>b.remove());
  const canvasPrev = document.getElementById('rutaCanvas');
  if (canvasPrev) canvasPrev.remove();

  (Array.isArray(edificios) ? edificios : []).forEach(e=>{
    const b=document.createElement('button');
    b.className='ubicacion-btn';
    b.textContent=e.nombre_edificio;

    const px = Number(e.pos_x) || 0;
    const py = Number(e.pos_y) || 0;
    const clampedX = Math.max(0, Math.min(MAP_W, px));
    const clampedY = Math.max(0, Math.min(MAP_H, py));

    b.style.left = clampedX + 'px';
    b.style.top = clampedY + 'px';
    b.dataset.id = String(e.id);
    b.onclick = () => manejarClick(Number(e.id));

    if(modo==='uno' && Number(e.id)===origenFijoId) b.classList.add('origen-fijo');
    if(Number(e.id)===seleccionOrigenId) b.classList.add('seleccionado-origen');
    if(Number(e.id)===seleccionDestinoId) b.classList.add('seleccionado-destino');

    cont.appendChild(b);
  });
}

function manejarClick(id){
  id = Number(id);
  if (Number.isNaN(id)) return;

  const ed = edificios.find(x=>Number(x.id)===id);
  const nombreEd = ed ? ed.nombre_edificio : 'Desconocido';

  if(modo==='uno'){
    if(id===origenFijoId){
      mostrarNotificacion('Este es el origen fijo','warning'); 
      return;
    }
    seleccionDestinoId = id;
    infoDestino.textContent = nombreEd;
    seleccionOrigenId = null;
    log(`Destino seleccionado: ${nombreEd} (ID: ${id})`);
  } else {
    if(!seleccionOrigenId){
      seleccionOrigenId = id;
      infoOrigen.textContent = nombreEd;
      log(`Origen seleccionado: ${nombreEd} (ID: ${id})`);
    } else if(!seleccionDestinoId){
      if(id===seleccionOrigenId){ 
        mostrarNotificacion('Elige otro edificio','warning'); 
        return; 
      }
      seleccionDestinoId = id;
      infoDestino.textContent = nombreEd;
      log(`Destino seleccionado: ${nombreEd} (ID: ${id})`);
    } else {
      seleccionOrigenId = seleccionDestinoId;
      seleccionDestinoId = id;
      const eo = edificios.find(x=>Number(x.id)===seleccionOrigenId);
      infoOrigen.textContent = eo ? eo.nombre_edificio : 'N/A';
      infoDestino.textContent = nombreEd;
      log(`Ruta actualizada: ${eo?.nombre_edificio} → ${nombreEd}`);
    }
  }
  renderizarBotones();
}

/* === NUEVO: detectar edificios cercanos a la ruta === */
function obtenerEdificiosCercanos(camino, distanciaMax=30){
  const cercanos = [];
  for(let i=0;i<camino.length-1;i++){
    const p1=camino[i], p2=camino[i+1];
    edificios.forEach(e=>{
      if(camino.some(c=>c.nombre===e.nombre_edificio)) return;
      const x0=Number(e.pos_x), y0=Number(e.pos_y);
      const dx=p2.x-p1.x, dy=p2.y-p1.y;
      const t=((x0-p1.x)*dx + (y0-p1.y)*dy)/(dx*dx+dy*dy);
      const tClamped=Math.max(0,Math.min(1,t));
      const xProj=p1.x+tClamped*dx, yProj=p1.y+tClamped*dy;
      const dist=Math.sqrt((x0-xProj)**2 + (y0-yProj)**2);
      if(dist<=distanciaMax) cercanos.push(e);
    });
  }
  const ids={};
  return cercanos.filter(e=>ids[e.id]?0:(ids[e.id]=1));
}

async function calcularRuta(){
  let origenId, destinoId;
  
  if (modo === 'uno') {
    origenId = Number(origenFijoId);
    destinoId = Number(seleccionDestinoId);
    if (!destinoId) {
      mostrarNotificacion('Selecciona destino', 'warning');
      return;
    }
  } else {
    origenId = Number(seleccionOrigenId);
    destinoId = Number(seleccionDestinoId);
    if (!origenId || !destinoId) {
      mostrarNotificacion('Selecciona origen y destino', 'warning');
      return;
    }
  }

  if (Number.isNaN(origenId) || Number.isNaN(destinoId)) {
    log('ERROR: IDs inválidos', 'error');
    return mostrarNotificacion('IDs inválidos', 'warning');
  }

  const origenNombre = edificios.find(e=>Number(e.id)===origenId)?.nombre_edificio;
  const destinoNombre = edificios.find(e=>Number(e.id)===destinoId)?.nombre_edificio;
  
  log(`🚀 Iniciando cálculo de ruta: ${origenNombre} (ID:${origenId}) → ${destinoNombre} (ID:${destinoId})`);

  const url = `/ruta/dijkstra/?origen=${encodeURIComponent(origenId)}&destino=${encodeURIComponent(destinoId)}`;
  log(`URL del request: ${url}`);

  try {
    log('Enviando petición al servidor...');
    const res = await fetch(url);
    log(`Respuesta recibida: Status ${res.status}`);
    
    if (!res.ok) {
      let errTxt = `Error servidor (${res.status})`;
      try { 
        const tmp = await res.json(); 
        errTxt += ` - ${JSON.stringify(tmp)}`; 
      } catch(e){}
      throw new Error(errTxt);
    }

    const data = await res.json();
    log('✅ Datos recibidos: ' + JSON.stringify(data));

    dibujarRuta(data.camino);

    const rutaJSON = generarJSONRuta(data.camino);
    document.getElementById('jsonRuta').textContent = JSON.stringify(rutaJSON, null, 2);
    log('✅ JSON generado y mostrado en pantalla', 'success');

    btnCalcular.style.display='none';
    btnNuevaRuta.style.display='inline-block';

  } catch(e){
    log(`ERROR: ${e.message}`,'error');
    mostrarNotificacion('Error al calcular ruta', 'error');
  }
}

/* === DIBUJAR RUTA === */
function dibujarRuta(camino){
  let canvas = document.getElementById('rutaCanvas');
  if(canvas) canvas.remove();
  canvas = document.createElement('canvas');
  canvas.id='rutaCanvas';
  canvas.width=MAP_W;
  canvas.height=MAP_H;
  canvas.style.position='absolute';
  canvas.style.top='0';
  canvas.style.left='0';
  canvas.style.zIndex='1';
  cont.appendChild(canvas);

  const ctx = canvas.getContext('2d');
  ctx.lineWidth=4;
  ctx.strokeStyle='rgba(231,76,60,0.9)';
  ctx.beginPath();
  camino.forEach((n,i)=>{
    const x=Number(n.pos_x), y=Number(n.pos_y);
    if(i===0) ctx.moveTo(x,y);
    else ctx.lineTo(x,y);
  });
  ctx.stroke();
}

/* === GENERAR JSON RUTA === */
function generarJSONRuta(camino){
  const puntos_asociados = [];
  const tipos = [];
  const direccion = [];
  const edificios_cercanos = [];
  const edificiosAgregados = new Set();

  for(let i=0;i<camino.length;i++){
    const nodo = camino[i];
    puntos_asociados.push(nodo.nom_nodo);
    tipos.push(nodo.tipo);

    // Direccion hacia siguiente nodo
    if(i < camino.length -1){
      const sig = camino[i+1];
      const dx = sig.pos_x - nodo.pos_x;
      const dy = sig.pos_y - nodo.pos_y;
      let dir='';
      if(Math.abs(dx)>Math.abs(dy)) dir = dx>0 ? 'derecha' : 'izquierda';
      else if(Math.abs(dy)>0) dir = dy>0 ? 'abajo' : 'arriba';
      else dir='mismo lugar';
      direccion.push(dir);
    }

    // Buscar edificios cercanos al nodo actual (exceptuando el nodo mismo)
    const cercanosNodo = [];
    edificios.forEach(e=>{
      if(e.nombre_edificio === nodo.nom_nodo) return; // ignorar el nodo
      if(edificiosAgregados.has(e.nombre_edificio)) return; // no repetir
      const dist = Math.sqrt((e.pos_x - nodo.pos_x)**2 + (e.pos_y - nodo.pos_y)**2);
      if(dist <= 30){ // distancia máxima para considerar "cercano"
        cercanosNodo.push({
          id: e.id,
          nombre: e.nombre_edificio,
          pos_x: e.pos_x,
          pos_y: e.pos_y,
          tipo: e.tipo || 'edificio'
        });
        edificiosAgregados.add(e.nombre_edificio);
      }
    });

    // Si no hay edificios cercanos, agregamos vacío
    edificios_cercanos.push(cercanosNodo.length>0 ? cercanosNodo : []);
  }

  // La dirección del último nodo se mantiene como vacío porque no hay siguiente nodo
  direccion.push('');

  return { puntos_asociados, tipos, direccion, edificios_cercanos };
}

/* === NUEVA BÚSQUEDA === */
function buscarNuevaRuta(){
  seleccionOrigenId = null;

  if(canvasPrev) canvasPrev.remove();
}

/* === NOTIFICACIONES === */
